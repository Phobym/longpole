# Связи между джобами и перетаскивание: план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Внутри стейджа джоба стоит сразу после соседа, которого ждала; при выборе джобы подсвечиваются её прямые зависимости и зависимые джобы; трейс сдвигается перетаскиванием мышью.

**Architecture:** Новая функция `orderByDeps` в `src/model.mjs` упорядочивает детей стейджа и группы и пишет в каждый узел `after`. `src/aggregate.mjs` строит `deps` агрегированных узлов и применяет ту же функцию. `src/template.html` выводит подпись `← родитель`, подсветку связей выбранного узла и сдвиг оси перетаскиванием.

**Tech Stack:** Node 22, ES-модули, `node:test`, vanilla JS в HTML-шаблоне. Зависимостей нет.

Спецификация: `docs/specs/2026-09-24-longpole-design.md`, раздел «Связи между джобами» и пункт «Управление» раздела «Интерфейс HTML».

## Global Constraints

- Без зависимостей в `package.json`; Node >= 22; ES-модули.
- `src/critical-path.mjs` не меняется и остаётся без `import`.
- Тексты для пользователя — на русском.
- Все значения из GitLab в HTML проходят через `esc()`.
- Комментарии в коде — только там, где причина неочевидна из кода.
- Спан получает поле `after: string | null` — id соседа-родителя по связям.
- Цвета подсветки: светлая тема `--dep-up: #0284c7; --dep-down: #16a34a;`, тёмная `--dep-up: #38bdf8; --dep-down: #4ade80;`.
- Порог начала перетаскивания — 4 px.

**Interfaces:**
- Produces:

Корни — по старту, под каждым узлом его дети по старту, рекурсивно. Узлы из цикла зависимостей попадают в конец по старту, каждый ровно один раз.
  - `export const byStart` — компаратор по `start`, `null` в конце (уже есть в файле, становится экспортом).
  - `span()` по умолчанию ставит `after: null`.
  - Спан `group` получает `deps` — объединение `deps` шардов без повторов.

```js
const leafIds = (s) => (s.kind === 'group' ? s.children.flatMap(leafIds) : [s.id])

export function orderByDeps(children) {
  const sorted = [...children].sort(byStart)
  const rank = new Map(sorted.map((s, i) => [s, i]))
  // зависимость может указывать на шард: владелец шарда — его группа среди соседей
  const owner = new Map()
  for (const s of sorted) for (const id of [s.id, ...leafIds(s)]) owner.set(id, s)

  const parentOf = new Map()
  for (const s of sorted) {
    let parent = null
    for (const id of s.deps) {
      const o = owner.get(id)
      if (!o || o === s) continue
      const later = (o.end ?? -Infinity) > (parent?.end ?? -Infinity)
      const tieEarlier = parent && (o.end ?? -Infinity) === (parent.end ?? -Infinity) && rank.get(o) < rank.get(parent)
      if (parent === null || later || tieEarlier) parent = o
    }
    parentOf.set(s, parent)
  }

  const kids = new Map(sorted.map((s) => [s, []]))
  const roots = []
  for (const s of sorted) (parentOf.get(s) ? kids.get(parentOf.get(s)) : roots).push(s)

  const out = []
  const seen = new Set()
  const visit = (s) => {
    if (seen.has(s)) return
    seen.add(s)
    out.push(s)
    kids.get(s).forEach(visit)
  }
  roots.forEach(visit)
  // узлы из цикла зависимостей не достижимы от корней
  sorted.forEach(visit)
  for (const s of out) s.after = parentOf.get(s)?.id ?? null
  return out
}
```

`src/aggregate.mjs` целиком:

```js
import { criticalPath } from './critical-path.mjs'
import { byStart, orderByDeps } from './model.mjs'

export function percentile(values, p) {
  if (values.length === 0) return null
  const sorted = [...values].sort((a, b) => a - b)
  return sorted[Math.max(0, Math.ceil((p / 100) * sorted.length) - 1)]
}

const stat = (values) => ({ p50: percentile(values, 50), p90: percentile(values, 90) })
const keyOf = (s) => (s.kind === 'pipeline' ? 'pipeline' : `${s.kind}:${s.name}`)
const LINKED = new Set(['job', 'bridge', 'group'])
const ORDERED_BY_DEPS = new Set(['stage', 'group'])

export function aggregate(trees) {
  const critical = trees.map((t) => new Set(criticalPath(t).ids))
  // id исходного спана в пайплайне → id агрегированного узла; нужен, чтобы перевести deps
  const aggIdOf = new Map()
  const linked = []

  // entries — один и тот же узел в разных пайплайнах: [{ span, tree }]
  const merge = (id, kind, name, entries) => {
    for (const e of entries) aggIdOf.set(`${e.tree}|${e.span.id}`, id)
    const groups = new Map()
    for (const e of entries) {
      for (const c of e.span.children) {
        const key = keyOf(c)
        if (!groups.has(key)) groups.set(key, { kind: c.kind, name: c.name, entries: [] })
        groups.get(key).entries.push({ span: c, tree: e.tree })
      }
    }
    const children = [...groups].map(([key, g]) => merge(`${id}/${key}`, g.kind, g.name, g.entries)).sort(byStart)

    const ran = entries.filter((e) => e.span.start != null && e.span.end != null)
    const stats = {
      present: ran.length,
      total: trees.length,
      start: stat(ran.map((e) => e.span.start)),
      end: stat(ran.map((e) => e.span.end)),
      duration: stat(ran.map((e) => e.span.end - e.span.start)),
      queued: stat(ran.map((e) => e.span.queued).filter((q) => q != null)),
      retries: ran.length ? ran.reduce((n, e) => n + e.span.attempts.length, 0) / ran.length : 0,
      critical: ran.filter((e) => critical[e.tree].has(e.span.id)).length / trees.length,
      samples: ran.map((e) => ({ tree: e.tree, start: e.span.start, end: e.span.end })),
    }
    const node = {
      id, kind, name,
      start: stats.start.p50, end: stats.end.p50, queued: stats.queued.p50,
      status: null, url: null, allowFailure: false, attempts: [], deps: [], after: null,
      children, stats,
    }
    if (LINKED.has(kind)) linked.push({ node, entries })
    return node
  }

  const root = merge('agg', 'pipeline', `${trees.length} пайплайнов`, trees.map((span, tree) => ({ span, tree })))

  for (const { node, entries } of linked) {
    const deps = new Set()
    for (const e of entries) {
      for (const dep of e.span.deps) {
        const aggId = aggIdOf.get(`${e.tree}|${dep}`)
        if (aggId && aggId !== node.id) deps.add(aggId)
      }
    }
    node.deps = [...deps]
  }
  const reorder = (s) => {
    if (ORDERED_BY_DEPS.has(s.kind)) s.children = orderByDeps(s.children)
    s.children.forEach(reorder)
  }
  reorder(root)
  return root
}
```

```js
const LINKABLE = new Set(['job', 'bridge', 'group'])
const leafIds = (s) => (s.kind === 'group' ? s.children.flatMap(leafIds) : [s.id])

// связанный узел внутри свёрнутой группы подсвечивается строкой группы
function visibleId(id) {
  let visible = id
  for (let p = state.parent.get(id); p; p = state.parent.get(p.id)) if (state.collapsed.has(p.id)) visible = p.id
  return visible
}

function relations() {
  const sel = state.selected && state.index.get(state.selected)
  if (!sel || !LINKABLE.has(sel.kind)) return { up: new Set(), down: new Set(), name: '' }
  const own = new Set([sel.id, ...leafIds(sel)])
  const up = new Set(sel.deps.map(visibleId))
  const down = new Set()
  for (const s of state.index.values()) {
    if (!own.has(s.id) && s.deps.some((d) => own.has(d))) down.add(visibleId(s.id))
  }
  up.delete(sel.id)
  down.delete(sel.id)
  return { up, down, name: sel.name }
}
```

- [ ] **Step 2: Обработчики**

1. В начале обработчика `document.addEventListener('click', (e) => {` первой строкой добавить:

```js
  // отпускание кнопки после перетаскивания порождает click — он не должен менять выделение

```

2. Перед этим обработчиком `click` добавить:

```js
const DRAG_THRESHOLD_PX = 4
let drag = null
let dragEndedAt = 0
let frame = 0
const redraw = () => {
  if (!frame) frame = requestAnimationFrame(() => { frame = 0; draw() })
}

$('#rows').addEventListener('mousedown', (e) => {
  const laneEl = e.target.closest('.lane')
  if (!laneEl || e.button !== 0) return
  e.preventDefault()
  drag = { x: e.clientX, view: state.view, width: laneEl.getBoundingClientRect().width, moved: false }
})

document.addEventListener('mousemove', (e) => {
  if (!drag) return
  const dx = e.clientX - drag.x
  if (!drag.moved && Math.abs(dx) < DRAG_THRESHOLD_PX) return
  drag.moved = true
  document.body.classList.add('dragging')
  const [a, b] = drag.view
  const d = (-dx / drag.width) * (b - a)
  state.view = [a + d, b + d]
  redraw()
})

document.addEventListener('mouseup', () => {
  if (drag?.moved) dragEndedAt = Date.now()
  drag = null
  document.body.classList.remove('dragging')
})
```
